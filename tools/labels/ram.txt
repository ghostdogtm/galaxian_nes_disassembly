004C curJoy                     ; buttons of the current player (synthesized by the demo AI in attract mode)
004E pauseFlag                  ; nonzero = game paused (Start)
004F playerState                ; GameHacking.org "invincible": 004F:00; 0 = ship in play, 1-$21 explosion, $21 = waiting to spawn
0050 explTileLo                 ; UpdateBgTiles: player explosion tile address
0051 explRow                    ; UpdateBgTiles: player explosion row 3..0
0053 curPlayer                 ; bit 0 = which player is playing (2P game)
0054 scoreAdd 5                 ; digits to add (scoreAdd+3 = hundreds, +4 = tens), see AddScore
0059 stage                      ; GameHacking.org: stage modifier
005A otherStage                 ; stage of the player who is waiting (2P game)
005B attackTimers 8             ; 8 counters (every 4 frames), each wraps at AttackTimerPeriods,X; DifficultyAttackMask picks which launch

0066 attackSide                 ; bit 0: LaunchFromLeft / LaunchFromRight

0068 diveSiren                  ; $32 when a diver launches: siren while aliens dive
006B songPtr 2                  ; MusicChannelUpdate: song data pointer (songTable,X*2 - $0840)
006D musRequest 4               ; per tune (0-3): start it; 0 extra life, 1 flagship hit, 2 alien hit, 3 game start.

0071 musChannel                 ; tune being updated (3..0)

0076 musTimer 4                 ; per tune: note duration counter; musTimer+3 (start tune) != 0 blocks the hum and SpawnPlayer
007E sndScratch 16              ; $7E-$8D cleared every frame by UpdateSound; $7E = 1 when a note starts
0080 sq1PeriodLo                ; SQ1 period written by MusicChannelUpdate

009C modeTimer                  ; general timer: title menu, story text, delay after a wave

; ---- $0100 page (above the stack, not cleared on warm reset)
0100 stackPage                ; stack page base, STA stackPage,X
0101 musicPlayerFlag            ; nonzero -> NMI runs MusicPlayer every frame. In REVA nothing sets it: ResetKeepSignature

0102 frameCounter 3             ; 24-bit, big-endian: $0102 high ... $0104 low
0105 resetSignature 16          ; copy of "HARUHISA UDAGAWA" (TCRF)
0115 resetCounter               ; TCRF: number of warm resets; $2D (45) / $0A (10) enable the music player
0116 scoreArea 26             ; $0116-$012F, set to $10 (blank) at power-on; AddScore uses scoreArea,Y
0117 scoreP1 6                 ; BCD digits, tile $10 = blank
0120 hiScore 6                  ; 5000 after power-on
0129 scoreP2 6
; $014E-$01FF songTable + tunes 1-3 and $0300-$067F chrData (texts): copied from CHR ROM at RESET,
; their labels are defined in src/chr.asm (tools/chrdata.py)

; ---- buffers
0200 oamBuffer 256              ; sprite page for OAM DMA ($4014 = $02)
0600 ramPage6                   ; RESET clears $0600-$06FF (the text copy then overwrites $0600-$067F)
068F demoTargetX                ; demo AI: X the ship walks to
0690 hudBuffer 44               ; HUD tiles for PPU $2074-$209F (two rows): 22 pairs queued one per frame by UpdateHudScores
06A0 hudScore 6                 ; score of the current player shown in the HUD (hudScore+9 = hiScore copy)
06B1 hudLives 3                 ; ship icons, lower half (tile $6F; upper half $6E at hudBuffer+1)
06B5 hudFlags 5                 ; stage flag icons (lower half; upper half at hudBuffer+5)

06B7 hudStageOnes
06BC textBuf 26                 ; one text row: PrintText fills it, WriteVramRow (NMI) copies it to textVram
06D6 textCharBias               ; subtracted from the characters by PrintText ($30 or 0)
06D7 textNtFlag                 ; PrintText: nonzero = second nametable ($28xx); also a step counter of IntroTextSequence

0700 diverEscort1 8             ; flagship: slot of the first escort

0720 diverColumn 8              ; formation column 0-9

0740 diverTargetX 8             ; X where the dive turns back toward the ship
0748 vramQueueTile2 16          ; NMI queue: second tile
0758 vramQueueTile1 16          ; NMI queue: first tile
0768 vramQueueLo 16             ; NMI queue: PPU address low
0778 vramQueueHi 16             ; NMI queue: PPU address high

0788 motionDir 24               ; bit 7 = negative; GameHacking.org "enemy shooting affected" 078D:00 = motionDir+5
079F formationDir               ; bit 7 = moving left (= motionDir+23), flipped by FormationBounce
07A0 motionDiv 24               ; divisor: speed = motionNum / motionDiv px per frame; 0 = object idle
07B8 motionNum 24               ; numerator, added to motionRem every frame
07D0 motionRem 24               ; remainder of the division (sub-pixel position)
07E8 motionStep 23              ; signed pixels per frame, computed by UpdateMotionSteps
07FF formationDx                ; formation drift per frame (= motionStep+23)
